const App = () => {
  const [input, setInput] = useState("");
  const [phase, setPhase] = useState("IDLE");
  const [step, setStep] = useState(0);
  const [logs, setLogs] = useState([]);
  const [failures, setFailures] = useState(0);
  const [results, setResults] = useState(null); 
  const [historyList, setHistoryList] = useState([]);
  const [eta, setEta] = useState(0);
  const [theme, setTheme] = useState("default");
  const [selectedProfile, setSelectedProfile] = useState(PROFILES[0].id);
  const [menuOpen, setMenuOpen] = useState(null);
  const [copyStatus, setCopyStatus] = useState(null);
  const [isAdmin, setIsAdmin] = useState(false);
  const [steps, setSteps] = useState(INITIAL_STEPS);
  
  // Workflow States
  const [workflows, setWorkflows] = useState([]);
  const [currentWorkflowName, setCurrentWorkflowName] = useState("");
  const [user, setUser] = useState(null);

  const scrollRef = useRef(null);
  const t = THEMES[theme];
  const currentProfile = PROFILES.find(p => p.id === selectedProfile);

  // Firebase Init
  useEffect(() => {
    const initAuth = async () => {
      if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
        await signInWithCustomToken(auth, __initial_auth_token);
      } else {
        await signInAnonymously(auth);
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return () => unsubscribe();
  }, []);

  // Sync Workflows
  useEffect(() => {
    if (!user) return;
    const q = collection(db, 'artifacts', appId, 'users', user.uid, 'workflows');
    const unsubscribe = onSnapshot(q, (snapshot) => {
      const wfs = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
      setWorkflows(wfs);
    }, (err) => console.error("Firestore Error:", err));
    return () => unsubscribe();
  }, [user]);

  useEffect(() => {
    if (scrollRef.current) scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
  }, [logs]);

  const callGemini = async (prompt, systemInstruction = "") => {
    const finalInstruction = `${currentProfile.instruction} ${systemInstruction}`;
    let delay = 1000;
    for (let i = 0; i < 5; i++) {
      try {
        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${PRIMARY_MODEL}:generateContent?key=${apiKey}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            contents: [{ parts: [{ text: prompt }] }],
            systemInstruction: { parts: [{ text: finalInstruction }] }
          })
        });
        const data = await response.json();
        if (!response.ok) throw new Error(data.error?.message || "API Error");
        return data.candidates[0].content.parts[0].text;
      } catch (err) {
        if (i === 4) throw err;
        await new Promise(res => setTimeout(res, delay));
        delay *= 2;
      }
    }
  };

  const addLog = (msg, type = "info") => {
    setLogs(prev => [...prev, { timestamp: new Date().toLocaleTimeString(), msg, type }]);
  };

  const updateEta = (currentStepIdx) => {
    const remaining = steps.slice(currentStepIdx).reduce((acc, s) => acc + s.baseEta, 0);
    setEta(remaining);
  };

  const handleRestart = (e) => {
    if (e) e.stopPropagation();
    setPhase("IDLE"); setStep(0); setLogs([]); setFailures(0); setResults(null); setEta(0);
    setMenuOpen(null);
    setSelectedProfile("architect");
    setTheme("default");
    setCurrentWorkflowName("");
  };

  const updateStepLabel = (id, newLabel) => {
    setSteps(prev => prev.map(s => s.id === id ? { ...s, label: newLabel } : s));
  };

  const processIdea = async () => {
    if (!input.trim() || phase === "PROCESSING") return;
    setPhase("PROCESSING"); setStep(0); setFailures(0); setLogs([]);
    addLog(`Refinement engine engaged. Profile: ${selectedProfile.toUpperCase()}.`, "system");

    try {
      let currentData = { originalInput: input };
      const runStep = async (idx, promptFn) => {
        setStep(idx + 1); updateEta(idx);
        addLog(steps[idx].label);
        return await promptFn();
      };

      currentData.expansion = await runStep(0, () => callGemini(`Expand concept (600-1000 words): ${input}`));
      currentData.initialRoast = await runStep(1, () => callGemini(`Identify weaknesses in: ${currentData.expansion}`));
      const v1 = await runStep(2, () => callGemini(`Respond SUCCESS/FAILURE. Prompt: ${input} Extension: ${currentData.expansion}`));
      if (!v1.toUpperCase().includes("SUCCESS")) throw new Error("Intent Divergence.");

      currentData.brainstorm60 = await runStep(3, () => callGemini(`Generate 60 modifications for: ${currentData.initialRoast}`));
      currentData.sortedIdeas = await runStep(4, () => callGemini(`Sort mods: ${currentData.brainstorm60}`));
      currentData.top30 = await runStep(5, () => callGemini(`Select top 30 from: ${currentData.sortedIdeas}`));
      const v2 = await runStep(6, () => callGemini(`Respond SUCCESS/FAILURE on mitigation check.`));
      if (!v2.toUpperCase().includes("SUCCESS")) throw new Error("Threshold Breach.");

      currentData.secondaryRoast = await runStep(7, () => callGemini(`Roast these 30 ideas: ${currentData.top30}`));
      currentData.concepts10 = await runStep(8, () => callGemini(`Combine into 10 modules: ${currentData.top30}`));
      const promptInstruction = `Synthesize Meta-Prompt (600-1200 words). Base: ${currentData.expansion} + ${currentData.concepts10}`;
      currentData.finalPromptBase = await runStep(9, () => callGemini(promptInstruction));
      currentData.versionA = await runStep(10, () => callGemini(`Tone: HS student. Base: ${currentData.finalPromptBase}`));
      currentData.versionB = await runStep(11, () => callGemini(`Tone: Masters grad. Base: ${currentData.finalPromptBase}`));
      
      addLog("Protocol Terminated Successfully.", "success");
      setResults(currentData);
      setHistoryList(prev => [{ id: Date.now(), input: input.slice(0, 25) + '...', timestamp: new Date().toLocaleTimeString() }, ...prev]);
      setPhase("COMPLETE");
      setEta(0);
    } catch (err) {
      addLog(`Operational Fault: ${err.message}`, "error");
      if (failures < 3) {
        setFailures(f => f + 1);
        addLog(`Auto-Recovery Attempt ${failures + 2}...`, "system");
        setTimeout(processIdea, 1500);
      } else {
        setPhase("ERROR");
      }
    }
  };

  const saveWorkflow = async () => {
    if (!user || !currentWorkflowName) return;
    const workflowData = {
      name: currentWorkflowName,
      input,
      steps,
      results,
      profile: selectedProfile,
      theme,
      timestamp: new Date().toISOString()
    };
    try {
      await addDoc(collection(db, 'artifacts', appId, 'users', user.uid, 'workflows'), workflowData);
      addLog(`Workflow '${currentWorkflowName}' stored in internal matrix.`, "success");
    } catch (err) {
      addLog(`Matrix Storage Fault: ${err.message}`, "error");
    }
  };

  const loadWorkflow = (wf) => {
    setCurrentWorkflowName(wf.name);
    setInput(wf.input);
    setSteps(wf.steps || INITIAL_STEPS);
    setResults(wf.results || null);
    setSelectedProfile(wf.profile || "architect");
    setTheme(wf.theme || "default");
    setPhase(wf.results ? "COMPLETE" : "IDLE");
    addLog(`Workflow '${wf.name}' restored. Integrity verified.`, "system");
    setMenuOpen(null);
  };

  const exportWorkflow = () => {
    if (!results) return;
    const md = `# Refiner Engine Workflow: ${currentWorkflowName || 'Unnamed'}\n\n` +
               `## Original Concept\n${input}\n\n` +
               `## Procedural Logic\n${steps.map(s => `- **${s.label.split('.')[0]}**: ${s.label}`).join('\n')}\n\n` +
               `## High School Synthesis\n${results.versionA}\n\n` +
               `## Master's Scholar Synthesis\n${results.versionB}\n\n` +
               `*Timestamp: ${new Date().toLocaleString()}*`;
    
    const blob = new Blob([md], { type: 'text/markdown' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `Workflow_${currentWorkflowName.replace(/\s+/g, '_') || 'Export'}.md`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
  };

  const importWorkflow = (e) => {
    const file = e.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (event) => {
      try {
        const data = JSON.parse(event.target.result);
        if (data.input) setInput(data.input);
        if (data.steps) setSteps(data.steps);
        if (data.name) setCurrentWorkflowName(data.name);
        addLog("External Workflow Template integrated.", "success");
      } catch (err) {
        addLog("Template Integrity Failure: Invalid JSON.", "error");
      }
    };
    reader.readAsText(file);
  };

  return (
    <div className={`min-h-screen ${t.bg} ${t.text} font-mono p-1 sm:p-2 transition-all duration-700 selection:bg-white/10 overflow-x-hidden`}>
      <div className="max-w-5xl mx-auto space-y-2 md:space-y-4 pb-12">
        
        {/* Header */}
        <header className={`${t.panel} border ${t.border} p-3 md:p-5 rounded-2xl md:rounded-[2rem] shadow-2xl relative z-50 flex justify-between items-start`}>
          <div className="absolute top-0 left-0 w-full h-[1px] md:h-[2px] bg-gradient-to-r from-transparent via-white/10 to-transparent" />
          
          <div className="flex items-start gap-4 md:gap-8 w-full md:w-auto">
            <EvilNecromancerAvatar />
            
            <div className="flex flex-col flex-1 relative">
              <h1 className={`text-base sm:text-xl md:text-2xl tracking-[0.4em] md:tracking-[1.2em] font-black uppercase italic leading-none relative ${t.accent} drip-text transform scale-y-90 origin-left whitespace-nowrap`}>
                Refiner Engine
              </h1>
              <div className={`text-sm md:text-base uppercase tracking-[0.1em] md:tracking-[0.2em] flex items-center gap-1.5 mt-1 font-bold ${t.subtext} transition-all duration-500`}>
                <Zap size={12} className="animate-pulse shrink-0" /> 
                {currentProfile.name} Mode Active
              </div>

              {/* Action Row */}
              <div className="flex items-center gap-2 mt-2 md:mt-3 shrink-0 relative">
                <Tooltip text="Purge Session State">
                  <button onClick={(e) => handleRestart(e)} className={`p-1.5 md:p-2 hover:bg-white/10 rounded-xl border ${t.border} transition-all shadow-lg active:scale-90`}>
                    <RotateCcw size={16} md:size={18} />
                  </button>
                </Tooltip>

                <div className="relative">
                  <button onClick={() => setMenuOpen(menuOpen === 'theme' ? null : 'theme')} className={`px-3 py-0.5 md:px-4 md:py-1 hover:bg-white/10 rounded-xl border ${t.border} transition-all flex items-center gap-2 text-sm md:text-base uppercase font-black tracking-widest whitespace-nowrap`}>
                    <Palette size={14} /> Theme
                  </button>
                  {menuOpen === 'theme' && (
                    <div className={`absolute top-full left-0 mt-1 w-48 md:w-56 ${t.panel} border ${t.border} rounded-xl shadow-2xl z-[100] overflow-hidden backdrop-blur-xl border-white/10 animate-in fade-in zoom-in-95 duration-200`}>
                      {Object.entries(THEMES).map(([k, themeData]) => (
                        <button key={k} onClick={() => { setTheme(k); setMenuOpen(null); }} className={`w-full text-left px-4 py-2 text-sm md:text-base uppercase font-black tracking-widest hover:bg-white/10 transition-colors flex items-center justify-between border-b border-white/5 last:border-0 ${theme === k ? t.accent : ''}`}>
                          <span>{themeData.name}</span>
                          <div className="flex gap-1 p-1 bg-black/20 rounded-full">
                            <div className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: themeData.preview[0] }} />
                            <div className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: themeData.preview[1] }} />
                          </div>
                        </button>
                      ))}
                    </div>
                  )}
                </div>

                <div className="relative">
                  <button onClick={() => setMenuOpen(menuOpen === 'model' ? null : 'model')} className={`px-3 py-0.5 md:px-4 md:py-1 hover:bg-white/10 rounded-xl border ${t.border} transition-all flex items-center gap-2 text-sm md:text-base uppercase font-black tracking-widest whitespace-nowrap`}>
                    <Box size={14} /> Profile
                  </button>
                  {menuOpen === 'model' && (
                    <div className={`absolute top-full left-0 mt-1 w-64 md:w-80 ${t.panel} border ${t.border} rounded-xl shadow-2xl z-[100] overflow-hidden backdrop-blur-xl border-white/10 animate-in fade-in zoom-in-95 duration-200`}>
                      {PROFILES.map(m => (
                        <button key={m.id} onClick={() => { setSelectedProfile(m.id); setMenuOpen(null); }} className={`w-full text-left px-5 py-4 border-b border-white/5 last:border-0 hover:bg-white/10 transition-colors ${selectedProfile === m.id ? t.accent : ''}`}>
                          <div className="text-sm md:text-base font-black uppercase flex items-center justify-between tracking-tighter">
                            {m.name}
                            {m.id === 'ghost' && <Skull size={14} className="animate-pulse text-red-600" />}
                          </div>
                          <div className="text-[10px] md:text-xs opacity-30 mt-1">{m.desc}</div>
                        </button>
                      ))}
                    </div>
                  )}
                </div>

                {/* Workflow Selector */}
                <div className="relative">
                  <button onClick={() => setMenuOpen(menuOpen === 'workflow' ? null : 'workflow')} className={`px-3 py-0.5 md:px-4 md:py-1 hover:bg-white/10 rounded-xl border ${t.border} transition-all flex items-center gap-2 text-sm md:text-base uppercase font-black tracking-widest whitespace-nowrap`}>
                    <FolderOpen size={14} /> Workflow
                  </button>
                  {menuOpen === 'workflow' && (
                    <div className={`absolute top-full left-0 mt-1 w-72 md:w-96 ${t.panel} border ${t.border} rounded-xl shadow-2xl z-[100] overflow-hidden backdrop-blur-xl border-white/10 animate-in fade-in zoom-in-95 duration-200`}>
                       <div className="p-4 border-b border-white/5 bg-white/5">
                          <input 
                            value={currentWorkflowName} 
                            onChange={(e) => setCurrentWorkflowName(e.target.value)}
                            placeholder="Workflow ID..."
                            className="w-full bg-black/40 border border-white/10 rounded-lg p-2 text-xs mb-2 outline-none focus:border-white/30"
                          />
                          <div className="flex gap-2">

                    </div>
                  </div>
                </div>
              )}

              {phase === "COMPLETE" && results && (
                <div className="space-y-6 animate-in fade-in zoom-in duration-1000 h-full flex flex-col pb-10">
                  <div className={`flex flex-col sm:flex-row items-center justify-between ${t.panel} p-6 rounded-2xl border ${t.border} shadow-2xl gap-4`}>

                      </button>
                    </Tooltip>
                  </div>

                  <div className="grid grid-cols-1 gap-6 flex-1">
                    <div className={`${t.panel} border ${t.border} rounded-2xl overflow-hidden flex flex-col shadow-2xl relative border-t-blue-600/30 border-t-2`}>
                      <div className="p-4 md:p-5 bg-white/5 border-b border-white/5 flex justify-between items-center">
                        <div className="flex items-center gap-3"><FileText className="text-blue-500" size={20} /><span className="text-xs md:text-sm font-black uppercase tracking-[0.1em]">Variant Alpha</span></div>
                        <div className="flex gap-3">
                          <button onClick={() => copyToClipboard(results.versionA, 'vA')} className="opacity-40 hover:opacity-100 transition-all p-2 bg-black/20 rounded-lg border border-white/10">{copyStatus === 'vA' ? <CheckCircle2 className="text-green-500" size={18} /> : <Copy size={18} />}</button>

                        </div>
                      </div>
                      <div className="p-6 md:p-10 text-base md:text-lg font-sans h-[350px] md:h-[450px] overflow-y-auto bg-black/20 leading-relaxed custom-scrollbar whitespace-pre-wrap select-text selection:bg-blue-500/20 font-medium">{results.versionA}</div>
                    </div>

                    <div className={`${t.panel} border ${t.border} rounded-2xl overflow-hidden flex flex-col shadow-2xl relative border-t-purple-600/30 border-t-2`}>

                        <div className="flex items-center gap-3"><Layers className="text-purple-500" size={20} /><span className="text-xs md:text-sm font-black uppercase tracking-[0.1em]">Variant Beta</span></div>

                          <button onClick={() => copyToClipboard(results.versionB, 'vB')} className="opacity-40 hover:opacity-100 transition-all p-2 bg-black/20 rounded-lg border border-white/10">{copyStatus === 'vB' ? <CheckCircle2 className="text-green-500" size={18} /> : <Copy size={18} />}</button>
                          <button onClick={() => downloadFile(results.versionB, "md")} className="opacity-40 hover:opacity-100 transition-all p-2 bg-black/20 rounded-lg border border-white/10"><FileJson size={18} /></button>
                        </div>
                      </div>
                      <div className="p-6 md:p-10 text-base md:text-lg font-sans h-[350px] md:h-[450px] overflow-y-auto bg-black/20 leading-relaxed custom-scrollbar whitespace-pre-wrap select-text selection:bg-purple-500/20 italic border-t border-white/5 font-medium">{results.versionB}</div>
                    </div>
                  </div>
                </div>
              )}
            </div>
          </div>
        )}

      </div>
      
      <style>{`
        .custom-scrollbar::-webkit-scrollbar { width: 3px; }
        @media (min-width: 768px) { .custom-scrollbar::-webkit-scrollbar { width: 6px; } }

        .drip-text::after {

              0% 0%, 100% 0%, 100% 75%, 98% 85%, 96% 75%, 94% 95%, 92% 75%, 90% 85%, 88% 75%, 86% 100%, 84% 75%, 82% 85%, 80% 75%,

    </div>
  );
